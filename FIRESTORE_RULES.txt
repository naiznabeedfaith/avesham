PUBLISH THESE RULES in Firebase Console → Firestore Database → Rules.
Review before launch. Orders are requests only; verify payments manually against your bank/UPI app.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() { return request.auth != null && request.auth.uid == "SajxZPyBwZbzzVJLXZWAvlmbJDR2"; }
    match /products/{productId} { allow read: if true; allow write: if isAdmin(); }
    match /settings/{settingId} { allow read: if true; allow write: if isAdmin(); }
    match /orders/{orderId} {
      allow create: if request.resource.data.keys().hasAll(['id','date','name','email','items','total','status'])
        && request.resource.data.id == orderId
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 120
        && request.resource.data.email is string && request.resource.data.email.size() <= 254
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 20
        && request.resource.data.total is number && request.resource.data.total >= 0
        && request.resource.data.status == 'Payment pending verification';
      allow read, update, delete: if isAdmin();
    }
    match /{document=**} { allow read, write: if false; }
  }
}
